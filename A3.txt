#include <iostream>
using namespace std;
#define SIZE 10
int hashFunction(int key) {
return key % SIZE;
}
void insert(int table[], int key, int method) {
int h = hashFunction(key);
for (int i = 0; i < SIZE; i++) {
int index;
if (method == 1)
index = (h + i) % SIZE;
else if (method == 2)
index = (h + i * i) % SIZE;
else
index = (h + i * (7 - key % 7)) % SIZE;
if (table[index] == -1) {
table[index] = key;
return;
}
}
cout << "Table is full\n";
}
void display(int table[]) {
for (int i = 0; i < SIZE; i++) {
cout << i << " : " << table[i] << endl;
}
}
int main() {
int linear[SIZE], quadratic[SIZE], doubleHash[SIZE];
int n, key;
for (int i = 0; i < SIZE; i++) { linear[i] = quadratic[i] = doubleHash[i] = -1;
}
cout << "Enter number of clients: ";
cin >> n;
cout << "Enter telephone numbers:\n";
for (int i = 0; i < n; i++) {
cin >> key;
insert(linear, key, 1);
insert(quadratic, key, 2);
insert(doubleHash, key, 3);
}
cout << "\nLinear Probing:\n";
display(linear);
cout << "\nQuadratic Probing:\n";
display(quadratic);
cout << "\nDouble Hashing:\n";
display(doubleHash);
return 0;
}
