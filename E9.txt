#include <iostream>
#include <vector>
using namespace std;
int main() {
int n, e;
cout << "Enter number of cities: ";
cin >> n;
vector<string> city(n);
for(int i = 0; i < n; i++) {
cout << "Enter city " << i + 1 << ": ";
cin >> city[i];
}
vector<vector<int>> adj(n);
cout << "Enter number of flights: ";
cin >> e;
cout << "Enter source destination:\n";
for(int i = 0; i < e; i++) {
int u, v;
cin >> u >> v;
adj[u-1].push_back(v-1);
adj[v-1].push_back(u-1);
}
// Display adjacency list
cout << "\nFlight Graph:\n";
for(int i = 0; i < n; i++) {
cout << city[i] << " -> ";
for(int j : adj[i])
cout << city[j] << " ";
cout << endl;
}
// Check connected
vector<bool> visited(n, false);
vector<int> q; q.push_back(0);
visited[0] = true;
for(int i = 0; i < q.size(); i++) {
int u = q[i];
for(int v : adj[u]) {
if(!visited[v]) {
visited[v] = true;
q.push_back(v);
}
}
}
bool connected = true;
for(bool x : visited)
if(!x) connected = false;
if(connected)
cout << "\nGraph is Connected";
else
cout << "\nGraph is Not Connected";
return 0;
}
